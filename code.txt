proc der(poly @f, poly @d){
    int i,j;
    poly @res=@f;
    list comp=factorize(@d);
    for(i=2;i<=size(comp[1]);i++){
        for(j=1;j<=comp[2][i];j++){
            @res=diff(@res,comp[1][i]);
        }
    }
return(@res);
}
proc derI(ideal @I, poly @d){
	ideal @res;
	for(int i=1;i<=size(@I);i++){
		if(typeof(@I[i])=="poly"){
			@res[i]=der(@I[i],@d);
		}
		else{
			@res[i]=@I[i];
		}
	}
return(@res);
}
proc maxId(int @k){
	def @r=basering; 
	ring @R=0,(x,y,z),dp;
	ideal MI=maxideal(@k);
	setring @r;
return(imap(@R,MI));
}
proc interpolationMatrixAfB(list L, int d, int m){
	ideal M=substitute(maxId(d),z,1);
    int range1=size(M);
	ideal M2=substitute(maxId(m-1),z,1);
	int range2=size(M2)+size(L);
	matrix im[range2][range1];
	for(int i=1;i<=size(M2);i++){
		im[i,1..range1]=substitute(derI(M,M2[i]),ideal(x,y),ideal(a,b));
	}
	for(i=size(M2)+1;i<=size(M2)+size(L);i++){
		im[i,1..range1]=substitute(M,ideal(x,y),ideal(L[i-size(M2)][1],L[i-size(M2)][2]));
	}
return(im);
}
